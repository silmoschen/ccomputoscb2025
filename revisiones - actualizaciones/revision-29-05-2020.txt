
/* Table: DETFACT, Owner: SYSDBA */

DROP TABLE DETFACT_HIST;

CREATE TABLE DETFACT_HIST
(
  PERIODO	 VARCHAR(7) CHARACTER SET WIN1251 NOT NULL,
  IDPROF	 VARCHAR(6) CHARACTER SET WIN1251 NOT NULL,
  CODOS	 VARCHAR(6) CHARACTER SET WIN1251 NOT NULL,
  ITEMS	 VARCHAR(3) CHARACTER SET WIN1251 NOT NULL,
  ORDEN	 VARCHAR(4) CHARACTER SET WIN1251 NOT NULL,
  CODPAC	 VARCHAR(20) CHARACTER SET WIN1251,
  NOMBRE	 VARCHAR(150) CHARACTER SET WIN1251,
  CODANALISIS	 VARCHAR(6) CHARACTER SET WIN1251,
  PROFIVA	 VARCHAR(1) CHARACTER SET WIN1251,
  OSIVA	 VARCHAR(1) CHARACTER SET WIN1251,
  REF1	 VARCHAR(10) CHARACTER SET WIN1251,
  RETIVA	 VARCHAR(2) CHARACTER SET WIN1251,
  MONTO	 DOUBLE PRECISION,
  IVA	 DOUBLE PRECISION,
  EXENTO	 DOUBLE PRECISION,
  NROAUDITORIA	 VARCHAR(50) CHARACTER SET WIN1251,
  NROAFILIADO	 VARCHAR(20) CHARACTER SET WIN1251,
  NROAUTORIZACION	 VARCHAR(20) CHARACTER SET WIN1251,
  FECHA	 VARCHAR(8) CHARACTER SET WIN1251,
  COSEGURO	 DOUBLE PRECISION,
CONSTRAINT PK_DETFACT_HIST PRIMARY KEY (PERIODO, IDPROF, CODOS, ITEMS, ORDEN)
);

/*  Index definitions for DETFACT */

CREATE INDEX DETFACT_HIST_RESUMENOS ON DETFACT(PERIODO, CODOS, IDPROF, ORDEN, ITEMS);
CREATE INDEX DETFACT_HIST_RESUMENPROF ON DETFACT(PERIODO, IDPROF, CODOS, ORDEN, ITEMS);
